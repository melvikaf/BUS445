capacities <- c(0.10, 0.20, 0.30)

evaluate_forest <- function(actual, probability, customer_id, model_name, fold_id) {

  outcome <- as.numeric(as.character(actual))

  positives <- sum(outcome == 1)
  negatives <- sum(outcome == 0)

  stopifnot(
    !anyNA(outcome),
    positives > 0,
    negatives > 0
  )

  # Calculate AUC
  auc <- (
    sum(rank(probability)[outcome == 1]) -
      positives * (positives + 1) / 2
  ) / (positives * negatives)

  # Rank customers by predicted churn probability
  ranked <- order(-probability, customer_id)

  # Evaluate different outreach capacities
  precision <- vapply(capacities, function(capacity) {
    n_contact <- floor(capacity * length(ranked))
    contacted <- head(ranked, n_contact)
    mean(outcome[contacted] == 1)
  }, numeric(1))

  # Calculate lift
  lift <- precision / mean(outcome == 1)

  data.frame(
    Model = model_name,
    Fold = fold_id,
    Capacity = capacities,
    AUC = auc,
    Precision = precision,
    Lift = lift
  )
}

Three-fold Cross Validation

This will train the model at each capacity for each fold. This may take a while to run.

fold_results <- list()
fold_models <- list()

for (fold_id in 1:3) {

  cat("Validation fold:", fold_id, "\n")

  # Split analysis and assessment data
  analysis <- training[training$cv_fold != fold_id, ]
  assessment <- training[training$cv_fold == fold_id, ]

  # Preprocess using analysis data only
  processed <- prepare_cell2cell_fold(
    analysis,
    assessment,
    inputs
  )

  analysis <- processed$analysis
  assessment <- processed$assessment

  # Fit and evaluate both Random Forest models
  fold_results[[fold_id]] <- do.call(
    rbind,
    lapply(names(model_terms), function(candidate) {

      predictors <- model_terms[[candidate]]

      # Set churn outcome as a factor
      outcome <- factor(
        as.character(analysis$churn_flag),
        levels = c("0", "1")
      )

      # Fit Random Forest
      set.seed(123 + fold_id)

      fit <- randomForest(
        x = analysis[, predictors, drop = FALSE],
        y = outcome,
        ntree = 500,
        importance = TRUE
      )

      # Predict churn probabilities
      probability <- predict(
        fit,
        newdata = assessment[, predictors, drop = FALSE],
        type = "prob"
      )[, "1"]

      stopifnot(
        length(probability) == nrow(assessment),
        all(is.finite(probability))
      )

      # Evaluate model
      results <- evaluate_forest(
        actual = assessment$churn_flag,
        probability = probability,
        customer_id = assessment$CustomerID,
        model_name = candidate,
        fold_id = fold_id
      )

      # Save fitted model
      fold_models[[paste(candidate, fold_id, sep = "_")]] <<- fit

      results
    })
  )
}

get_churn_matrix <- function(actual, probability, customer_id, capacity) {

  outcome <- as.numeric(as.character(actual))

  n <- length(probability)
  n_contact <- floor(n * capacity)

  # Rank customers by predicted probability
  ranked <- order(-probability, customer_id)

  # Select the top customers
  predicted <- rep(0, n)
  predicted[head(ranked, n_contact)] <- 1

  # Generate confusion matrix
  table(
    Actual = factor(outcome, levels = c(0, 1)),
    Predicted = factor(predicted, levels = c(0, 1))
  )
}

Adjusts the model’s predicted churn probabilities to reflect an assumed real-world churn rate.

adjust_probabilities <- function(probability, sample_rate, target_rate) {

  sample_odds <- probability / (1 - probability)

  adjusted_odds <- sample_odds *
    (target_rate / (1 - target_rate)) /
    (sample_rate / (1 - sample_rate))

  adjusted_probability <- adjusted_odds / (1 + adjusted_odds)

  adjusted_probability
}

Calculate Net Revenue for each combination of the assumptions.

# Create all combinations of assumptions
scenarios <- expand.grid(
  Churn_Rate = churn_rates,
  Retention_Success_Rate = retention_success_rates,
  Contact_Cost = contact_costs,
  Offer_Cost = offer_costs
)

# Customer monthly revenue
monthly_revenue <- as.numeric(final_test$MonthlyRevenue)

# Number of customers to contact
n_contact <- floor(nrow(final_test) * outreach_capacity)

# Calculate revenue for each scenario
revenue_results_rf <- do.call(
  rbind,
  lapply(seq_len(nrow(scenarios)), function(i) {

    scenario <- scenarios[i, ]

    # Adjust probabilities to the assumed real-world churn rate
    adjusted_probability <- adjust_probabilities(
      probability = test_probability,
      sample_rate = sample_churn_rate,
      target_rate = scenario$Churn_Rate
    )

    # Calculate monthly revenue at risk
    value_at_risk <- adjusted_probability * monthly_revenue

    # Rank customers by revenue at risk
    ranked_customers <- order(
      -value_at_risk,
      final_test$CustomerID
    )

    # Select customers within outreach capacity
    contacted <- head(ranked_customers, n_contact)

    # Expected number of at-risk customers contacted
    expected_at_risk <- sum(
      adjusted_probability[contacted]
    )

    # Expected number of customers retained
    expected_retained <- expected_at_risk *
      scenario$Retention_Success_Rate

    # Expected 12-month revenue saved
    revenue_saved <- sum(
      adjusted_probability[contacted] *
        scenario$Retention_Success_Rate *
        monthly_revenue[contacted] *
        projection_months
    )

    # Total campaign cost
    campaign_cost <- n_contact * (
      scenario$Contact_Cost + scenario$Offer_Cost
    )
    
    # Net revenue
    net_revenue <- revenue_saved - campaign_cost

    data.frame(
      Churn_Rate = scenario$Churn_Rate,
      Retention_Success_Rate = scenario$Retention_Success_Rate,
      Contact_Cost = scenario$Contact_Cost,
      Offer_Cost = scenario$Offer_Cost,
      Customers_Contacted = n_contact,
      Expected_At_Risk = expected_at_risk,
      Expected_Retained = expected_retained,
      Revenue_Saved = revenue_saved,
      Campaign_Cost = campaign_cost,
      Net_Revenue = net_revenue
    )
  })
)
